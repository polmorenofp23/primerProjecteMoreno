-- Insert extra ingredients based on categories
-- This will automatically add all ingredients from specified categories as extras
INSERT INTO product_ingredient (id_product, id_ingredient, grams_per_portion, portion_price, is_default)
SELECT 
    pec.id_product,
    i.id_ingredient,
    pec.grams_per_portion,
    pec.portion_price,
    0 AS is_default  -- These are extra ingredients, not default
FROM 
    temp_product_extra_categories pec
INNER JOIN 
    ingredient i ON i.category = pec.ingredient_category
WHERE 
    -- Avoid duplicates: don't add ingredient if it's already in the product (as default)
    NOT EXISTS (
        SELECT 1 
        FROM product_ingredient pi 
        WHERE pi.id_product = pec.id_product 
          AND pi.id_ingredient = i.id_ingredient
    )
ORDER BY 
    pec.id_product, i.id_ingredient;

-- Clean up temporary table
DROP TEMPORARY TABLE IF EXISTS temp_product_extra_categories;
